Planificateur

<poad-scheduler>

Les ressources en lignes, le temps à l’horizontale : qui fait quoi, quand. Une tâche affectée à plusieurs ressources dessine une barre sur chaque ligne, les chevauchements s’empilent, et, si on le permet, on déplace, redimensionne, réaffecte et crée à la souris, au doigt ou au clavier.

import {
  PoadSchedulerComponent, PoadSegmentedComponent, type PoadPlan, type PoadSchedulerAssignEvent, type PoadSchedulerChangeEvent, type PoadSchedulerCreateEvent,
  type PoadSchedulerScale, type PoadSchedulerScheduleEvent, type PoadSegmentOption,
} from '@poadio/ui-angular';

const PLAN: PoadPlan = {
  resources: [
    { id: 'field', label: 'Field team' },
    { id: 'claire', label: 'Claire Fontaine', parent: 'field', color: '#2f6feb' },
    { id: 'marc', label: 'Marc Delorme', parent: 'field', color: '#a371f7' },
    { id: 'van', label: 'Inspection van', color: '#0f9d58' },
  ],
  tasks: [
    { id: 's1', label: 'Site survey', start: '2026-09-14T09:00', end: '2026-09-15T17:00', assignments: [{ resource: 'claire' }] },
    { id: 's2', label: 'Bridge inspection', start: '2026-09-16T08:00', end: '2026-09-17T12:00', assignments: [{ resource: 'claire' }, { resource: 'van' }] },
    { id: 's3', label: 'Design review', start: '2026-09-14T09:00', end: '2026-09-14T17:00', assignments: [{ resource: 'marc' }] },
    { id: 's4', label: 'Report writing', start: '2026-09-15T09:00', end: '2026-09-17T17:00', assignments: [{ resource: 'marc' }] },
    { id: 's5', label: 'Servicing', start: '2026-09-21T08:00', end: '2026-09-21T17:00', assignments: [{ resource: 'van' }], locked: true },
    { id: 's6', label: 'Load test', duration: '1d' },
  ],
};
const hm = (d: Date) => `${d.getDate()}/${d.getHours()}:${String(d.getMinutes()).padStart(2, '0')}`;

@Component({
  selector: 'app-scheduler-demo',
  imports: [PoadSchedulerComponent, PoadSegmentedComponent],
  template: `
    <poad-segmented [(value)]="scale" [options]="scales" ariaLabel="Scale" size="sm" />
    <poad-scheduler class="demo-scheduler" [plan]="plan()" [scale]="scale()" [date]="date" [today]="today" editable ariaLabel="Team schedule"
      (eventMove)="change($event)" (eventResize)="change($event)" (eventAssign)="assign($event)" (eventCreate)="create($event)" (eventSchedule)="schedule($event)" />
    <code>{{ last() }}</code>
  `,
})
export default class SchedulerDemo {
  readonly scales: PoadSegmentOption[] = [{ label: 'Day', value: 'day' }, { label: 'Week', value: 'week' }, { label: 'Month', value: 'month' }];
  readonly scale = signal<PoadSchedulerScale>('week');
  readonly date = new Date(2026, 8, 14);
  readonly today = new Date(2026, 8, 15, 11);
  readonly plan = signal(PLAN);
  readonly last = signal('last=—');
  private nextId = 7;
  /** The scheduler never writes: the application applies the change to its own document. */
  change(e: PoadSchedulerChangeEvent): void {
    this.plan.update((p) => ({ ...p, tasks: p.tasks!.map((t) => (t.id === e.occurrence.taskId ? { ...t, start: e.start, end: e.end } : t)) }));
    this.last.set(`last=${e.occurrence.task.label} ${hm(e.start)}`);
  }
  assign(e: PoadSchedulerAssignEvent): void {
    this.plan.update((p) => ({ ...p, tasks: p.tasks!.map((t) => (t.id === e.occurrence.taskId ? { ...t, assignments: t.assignments!.map((a, i) => (i === e.assignmentIndex ? { ...a, resource: e.to } : a)) } : t)) }));
    this.last.set(`last=${e.occurrence.task.label} → ${e.to}`);
  }
  create(e: PoadSchedulerCreateEvent): void {
    this.plan.update((p) => ({ ...p, tasks: [...p.tasks!, { id: `s${this.nextId++}`, label: 'New task', start: e.start, end: e.end, assignments: [{ resource: e.resource }] }] }));
    this.last.set(`last=created ${e.resource} ${hm(e.start)}`);
  }
  schedule(e: PoadSchedulerScheduleEvent): void {
    this.plan.update((p) => ({ ...p, tasks: p.tasks!.map((t) => (t.id === e.task.id ? { ...t, start: e.start, end: e.end, duration: undefined, assignments: [{ resource: e.resource }] } : t)) }));
    this.last.set(`last=${e.task.label} → ${e.resource} ${hm(e.start)}`);
  }
}

    update(m.occurrence.taskId, () => ({ start: m.start, end: m.end }));
    out.textContent = `last=${m.occurrence.task.label} ${hm(m.start)}`;
  };
  sch.addEventListener('eventmove', change);
  sch.addEventListener('eventresize', change);
  sch.addEventListener('eventassign', (e) => {
    const m = e.detail;
    update(m.occurrence.taskId, (t) => ({ assignments: t.assignments.map((a, i) => (i === m.assignmentIndex ? { ...a, resource: m.to } : a)) }));
    out.textContent = `last=${m.occurrence.task.label} → ${m.to}`;
  });
  sch.addEventListener('eventcreate', (e) => {
    sch.plan = { ...sch.plan, tasks: [...sch.plan.tasks, { id: `s${nextId++}`, label: 'New task', start: e.detail.start, end: e.detail.end, assignments: [{ resource: e.detail.resource }] }] };

  });
  sch.addEventListener('eventschedule', (e) => {
    const m = e.detail;
    update(m.task.id, () => ({ start: m.start, end: m.end, duration: undefined, assignments: [{ resource: m.resource }] }));
    out.textContent = `last=${m.task.label} → ${m.resource} ${hm(m.start)}`;
  });

Chargement de la démo…

La démo ne s’est pas chargée. Une nouvelle version du site vient peut-être d’être publiée.

La démo tourne ici dans une vraie fenêtre de 390 px : ses règles « mobile » s’appliquent (feuilles basses, cibles tactiles). Tirez le coin bas-droit pour changer la largeur.

Vous êtes déjà sur un petit écran : la démo ci-dessus s’y comporte comme sur un téléphone. Pour la voir seule, en plein écran :

Ouvrir la démo en plein écran

À quoi ça sert

Pour répondre à « qui est libre jeudi ? ». Pour « que se passe-t-il le 17 ? », le calendrier ; pour des tâches liées dans la durée, le Gantt. Le planificateur n’écrit jamais : il émet l’intention (nouvelles dates, nouvelle ressource) et garde l’affichage à jour le temps que l’application l’applique. Une tâche sans dates ou sans ressource part dans la zone « non planifié », comptée, d’où on la pose sur une ligne.

Accessibilité

  • Région nommée ; chaque barre est un bouton dont le nom dit la tâche, sa période et « multi-ressource » s’il y a lieu ; la bulle ajoute le motif d’un verrou (tâche verrouillée, occurrence de série).
  • Édition active : chaque barre manipulable est décrite par l’aide clavier (aria-describedby) ; chaque geste est annoncé dans une région aria-live, avec la date d’arrivée ou la ressource dans la langue (« “Site survey” moved to Tuesday 15 September », « “Site survey” assigned to Marc Delorme ») ; le focus suit la barre, même quand elle change de ligne.
  • Dates en heure locale, changements d’heure compris ; noms de jours dans la langue. Gestes aimantés : au jour aux échelles semaine et mois, au pas (snap) à l’échelle jour. Un déplacement vertical ne change que la ressource, jamais les dates : une intention à la fois.
  • Sous le seuil mobile, auto passe en liste par ressource : une grille de quatorze jours à 390 px ne montre plus rien.
