'use client';

import { useState } from 'react';
import { PoadButton, PoadProgress } from '@poadio/ui-react';

export default function ProgressDemo() {
  const [build, setBuild] = useState(40);
  return (
    <>
      <div className="demo-col">
        <PoadProgress value={build} label="Build" showValue />
        <PoadProgress value={82} severity="warning" size="sm" striped aria-label="Disk usage" />
        <PoadProgress indeterminate aria-label="Upload" />

      </div>
    </>
  );
}

Accessibilité

Comment changer

Dans la feuille globale de l’application, après les feuilles de @poadio/ui-core : sur :root pour toute l’application, ou sur un conteneur pour une zone seulement. Une couche flottante (menu, liste, dialogue) peut être rendue hors de son conteneur : pour elle, préférez :root.

src/styles.css
/* Toute l’application */
:root {
  --poad-accent: #0b7a5c;        /* Couleur d’accent : bouton principal, case cochée, sélection */
  --poad-font-size-sm: 0.8125rem; /* Petit texte : aides, compteurs, badges */
}

/* En thème sombre seulement */
.theme-dark {
  --poad-accent: #3fcf9f;        /* Couleur d’accent : bouton principal, case cochée, sélection */
}

/* Une zone seulement */
.checkout {
  --poad-font-size-sm: 0.875rem; /* Petit texte : aides, compteurs, badges */
}
