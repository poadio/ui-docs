    <code>{{ last() }}</code>
  `,
})
export default class CalendarDemo {
  readonly views: PoadSegmentOption[] = [{ label: 'Month', value: 'month' }, { label: 'Week', value: 'week' }, { label: 'Agenda', value: 'agenda' }];
  readonly view = signal<PoadCalendarView>('week');

    this.plan.update((p) => ({ ...p, tasks: [...p.tasks!, { id: `e${this.nextId++}`, label: 'New event', bucket: 'office', start: e.start, end: e.end }] }));

  }
}

À quoi ça sert

Pour répondre à « que se passe-t-il le 17 ? ». Pour « qui est libre jeudi ? » (des ressources en lignes), le planificateur ; pour des tâches liées dans la durée, le Gantt. Le calendrier n’écrit jamais : il émet l’intention (nouvelles dates) et garde l’affichage à jour le temps que l’application l’applique. Une occurrence de série (règle rrule) est montrée verrouillée : l’édition des séries viendra plus tard.

Accessibilité

  • Région nommée ; chaque barre est un bouton dont le nom dit la tâche, sa période et « multi-ressource » s’il y a lieu ; la bulle ajoute le motif d’un verrou (tâche verrouillée, occurrence de série).
  • Édition active : chaque barre manipulable est décrite par l’aide clavier (aria-describedby) ; chaque geste est annoncé dans une région aria-live, avec la date d’arrivée ou la ressource dans la langue (« “Site survey” moved to Tuesday 15 September », « “Site survey” assigned to Marc Delorme ») ; le focus suit la barre, même quand elle change de ligne.
  • Dates en heure locale, changements d’heure compris ; noms de jours dans la langue. Gestes aimantés : au jour aux échelles semaine et mois, au pas (snap) à l’échelle jour. Un déplacement vertical ne change que la ressource, jamais les dates : une intention à la fois.
  • Sous le seuil mobile, auto passe en liste par ressource : une grille de quatorze jours à 390 px ne montre plus rien.
