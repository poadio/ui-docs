import { Component, computed, signal } from '@angular/core';
import { PoadColumnComponent, PoadSegmentedComponent, PoadTableComponent, tableEditApply, type PoadEditCompleteEvent, type PoadRowEditEvent, type PoadSegmentOption } from '@poadio/ui-angular';

interface Service { id: number; name: string; env: string; pods: number; active: boolean }
type Mode = 'cell' | 'row-inline' | 'row';

@Component({
  selector: 'app-table-edit-demo',
  imports: [PoadTableComponent, PoadColumnComponent, PoadSegmentedComponent],
  template: `
    <poad-segmented [(value)]="mode" [options]="modes" ariaLabel="Edit mode" size="sm" />
    <poad-table [value]="services()" dataKey="id" title="Services" [editMode]="editMode()" rowActionsAlwaysVisible (editComplete)="save($event)" (rowEditSave)="saveRow($event)">
      <poad-column field="name" header="Service" editable required [minLength]="3" />
      <poad-column field="env" header="Env" width="110px" editable editor="select" [editorOptions]="envs" />
      <poad-column field="pods" header="Pods" dataType="number" width="80px" editable editor="number" [min]="0" [max]="20" />
      <poad-column field="active" header="Active" dataType="boolean" width="90px" editable editor="checkbox" />
    </poad-table>
    <code>last={{ last() }}</code>
  `,
})
export default class TableEditDemo {
  /** Les trois modes : une cellule, toute la ligne en place, ou la ligne dans une boîte de dialogue. */
  readonly modes: PoadSegmentOption[] = [{ label: 'Cell', value: 'cell' }, { label: 'Row', value: 'row-inline' }, { label: 'Dialog', value: 'row' }];
  readonly mode = signal<unknown>('cell');
  readonly editMode = computed(() => this.mode() as Mode);
  readonly envs = ['recette', 'neo', 'prod'].map((e) => ({ label: e, value: e }));
  readonly services = signal<Service[]>([
    { id: 1, name: 'web-01', env: 'prod', pods: 4, active: true },
    { id: 2, name: 'web-02', env: 'prod', pods: 4, active: true },
    { id: 3, name: 'api', env: 'neo', pods: 2, active: true },
    { id: 4, name: 'worker', env: 'recette', pods: 1, active: false },
    { id: 5, name: 'cron', env: 'recette', pods: 1, active: true },
  ]);
  readonly last = signal('—');
  /** La table ne modifie pas les lignes : l'application écrit la valeur validée. */
  save(e: PoadEditCompleteEvent<Service>): void {
    this.services.set(tableEditApply(this.services(), e.row, { [e.column.field()]: e.value }));
    this.last.set(`${e.column.field()}=${e.value}`);
  }
  /** Modes ligne : seuls les champs changés arrivent. */
  saveRow(e: PoadRowEditEvent<Service>): void {
    this.services.set(tableEditApply(this.services(), e.row, e.changes));
    this.last.set(Object.entries(e.changes).map(([k, v]) => `${k}=${v}`).join(','));
  }
}

À quoi ça sert

Pour des corrections rapides sur des données déjà affichées (nombre de pods, environnement, statut). editMode="cell" : double-clic, Entrée ou F2 ouvrent l’éditeur de la cellule, Entrée valide, Échap annule, Tab passe à la cellule éditable suivante. editMode="row-inline" : un bouton Modifier par ligne ouvre toute la ligne, Entrée ou ✓ enregistre, Échap ou ✕ annule. Par colonne : editable, editor (text, number, select, checkbox, date, textarea), editorOptions, contraintes required, min / max, minLength / maxLength, et validator pour vos règles ; un éditeur de l’application remplace l’éditeur intégré (gabarit poadEditorTemplate, editorCell, slot #editor, renderEditor). La table ne modifie pas vos données : écrivez changes de editComplete / rowEditSave dans value (tableEditApply) ; en Vue, v-model:value le fait ; le composant web écrit value lui-même, sauf si vous annulez l’évènement. Pas d’édition en défilement virtuel : l’éditeur et son focus disparaîtraient avec sa ligne quand elle sort de la tranche rendue, et le passage à la cellule suivante viserait des cellules absentes du DOM ; pour éditer une longue liste, utilisez la pagination. editMode="row" : le bouton « Modifier la ligne » ouvre la ligne dans une boîte de dialogue de la bibliothèque, un champ par colonne éditable, validé à la frappe ; « Enregistrer » s’active au premier changement et n’envoie que les champs changés. C’est le bon choix pour un formulaire long ou des champs larges (titre, taille et colonnes du formulaire : page « Édition en boîte de dialogue »). La démo passe d’un mode à l’autre.

Accessibilité

  • En mode cellule, la table est une grille (role="grid") ; chaque cellule éditable est décrite par l’aide « Entrée ou F2 pour modifier » (aria-describedby).
  • L’éditeur est nommé par l’en-tête de sa colonne ; une valeur invalide pose aria-invalid et affiche son message (role="alert"), tiré du catalogue (table.editRequired, table.editMax…).
  • L’éditeur « liste » est le menu déroulant de la bibliothèque, jamais une liste native. Après validation ou annulation, le focus revient toujours sur la cellule, jamais sur la page.
  • En mode boîte de dialogue, la boîte est modale (role="dialog", focus piégé), nommée par son titre ; ses champs sont nommés par les en-têtes et portent aria-invalid et leur message en cas d’erreur ; fermée, elle rend le focus au bouton de la ligne.

À quoi ça sert

Quand une ligne a trop de champs pour s’éditer en place, ou des champs longs (une description). editMode="row" ajoute le bouton « Modifier la ligne » dans la colonne d’actions ; la boîte est celle de la bibliothèque (modale, focus piégé, rendu au bouton à la fermeture). Comme les autres actions de ligne, ce bouton apparaît au survol et au focus ; rowActionsAlwaysVisible le laisse toujours visible, ce que fait la démo. Ses champs reprennent les éditeurs et les contraintes des colonnes (editor, required, min / max, minLength / maxLength, validator) : les mêmes que l’édition dans la table. rowEditDialogHeader donne le titre, texte ou fonction de la ligne (défaut : « Modifier ») ; rowEditDialogSize sa taille (sm, md, lg) ; rowEditDialogCols le nombre de colonnes du formulaire (1 ou 2 ; une zone de texte prend toute la largeur). Seuls les champs changés partent dans rowEditSave (changes) ; « Annuler », Échap ou le fond émettent rowEditCancel. Comme pour les autres modes, la table ne modifie pas vos données (voir « Édition dans la table »).

Accessibilité

  • La boîte est un role="dialog" modal nommé par son titre ; chaque champ est nommé par l’en-tête de sa colonne.
  • Une valeur invalide pose aria-invalid et affiche son message sous le champ, tiré du catalogue (table.editRequired, table.editMax…) ; « Enregistrer » ne part pas tant qu’une erreur reste.

À quoi ça sert

Quand l’ordre est une donnée que l’utilisateur décide (étapes d’une mise en production, priorités d’un lot, ordre d’affichage). reorderableRows ajoute une colonne de poignées ; la table n’applique pas l’ordre elle-même : rendez rows de l’évènement rowReorder dans value (en Vue, v-model:value ; le composant web range value lui-même, sauf si vous annulez rowreorder). dragIndex et dropIndex sont des places dans value. Avec la pagination, une ligne change de place dans sa page ; la sélection suit les lignes. rowDragDisabled exclut des lignes ; rowLabel nomme une ligne dans les annonces et sur sa poignée (« Smoke tests saisie, position 5 sur 6 », « Déplacer Smoke tests ») au lieu de sa place. Pas de poignée quand l’ordre est imposé ou inconnu : sous un tri, avec des groupes ou un arbre, en données serveur (lazy), en cartes, et en défilement virtuel (la ligne saisie sortirait de la tranche rendue pendant le défilement automatique ; pour de longues listes, utilisez la pagination).

À quoi ça sert

Quand découper en pages gêne (journal, inventaire qu’on parcourt d’un geste) et que les lignes se comptent en milliers. virtualScroll active la tranche ; donnez une hauteur à la zone qui défile (scrollHeight) et une hauteur de ligne fixe (virtualRowHeight, défaut 42 px, 34 en dense ; les cellules la tiennent, sur une ligne). virtualBuffer (défaut 10) rend quelques lignes de plus de chaque côté. Le paginateur est coupé ; en cartes (responsive="stack"), la hauteur n’est plus fixe et le virtuel s’efface. scrollToIndex et scrollToRow amènent une ligne en haut de la vue. Une ligne de détail dépliée n’est pas comptée dans la hauteur : préférez-la sans virtuel.
