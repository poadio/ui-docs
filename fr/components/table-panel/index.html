Panneau de table

<poad-table-panel>

Un panneau latéral à côté d’une table, avec un rail d’onglets verticaux : Colonnes (montrer, masquer, chercher une colonne) et Filtres (recherche globale et un éditeur par colonne filtrable : valeurs à cocher, oui / non, opérateur et nombre, date ou texte).

<poad-table-panel [table]="t"> à côté de <poad-table #t> : le panneau lit et change la table elle-même.

import { Component, signal } from '@angular/core';
import { PoadColumnComponent, PoadTableComponent, PoadTablePanelComponent, type PoadFilters } from '@poadio/ui-angular';

interface Project { id: number; name: string; env: string; pods: number; owner: string }

@Component({
  selector: 'app-table-panel-demo',
  imports: [PoadTableComponent, PoadColumnComponent, PoadTablePanelComponent],
  template: `
    <div class="demo-wide" style="display: flex; align-items: stretch">
      <poad-table #t [value]="projects" dataKey="id" title="Projects" [showFilterRow]="false" style="flex: 1; min-width: 0">
        <poad-column field="name" header="Name" sortable filterable />
        <poad-column field="env" header="Env" filterable filterType="select" [filterOptions]="envs" />
        <poad-column field="pods" header="Pods" dataType="number" sortable filterable />
        <poad-column field="owner" header="Owner" />
      </poad-table>
      <poad-table-panel [table]="t" [tabs]="['columns', 'filters']" (filtersChange)="changed($event)" />
    </div>
    <code>filters={{ active() }}</code>
  `,
})
export default class TablePanelDemo {
  readonly projects: Project[] = [
    { id: 1, name: 'gestion', env: 'recette', pods: 4, owner: 'Hélène' }, { id: 2, name: 'inventaire', env: 'neo', pods: 0, owner: 'Zoé' },
    { id: 3, name: 'demo2', env: 'neo', pods: 2, owner: 'Noël' }, { id: 4, name: 'best', env: 'recette', pods: 12, owner: 'Hélène' },
    { id: 5, name: 'atlas', env: 'neo', pods: 7, owner: 'Zoé' },
  ];
  readonly envs = [{ label: 'recette', value: 'recette' }, { label: 'neo', value: 'neo' }];
  readonly active = signal('none');
  changed(filters: PoadFilters): void {
    this.active.set(Object.keys(filters).filter((k) => filters[k]).join(',') || 'none');
  }
}

Next.js emploie le même paquet que React : la démo est rendue par @poadio/ui-react. Le code montre l'intégration App Router — un composant client ('use client') pour l'état, rendu par une page serveur.

// app/table-panel-demo.tsx — composant client (état, gestionnaires)
'use client';

import { useState } from 'react';
import { PoadTable, PoadTablePanel, type PoadTableColumn, type TableFilters } from '@poadio/ui-react';

interface Project { id: number; name: string; env: string; pods: number; owner: string }
const projects: Project[] = [
  { id: 1, name: 'gestion', env: 'recette', pods: 4, owner: 'Hélène' }, { id: 2, name: 'inventaire', env: 'neo', pods: 0, owner: 'Zoé' },
  { id: 3, name: 'demo2', env: 'neo', pods: 2, owner: 'Noël' }, { id: 4, name: 'best', env: 'recette', pods: 12, owner: 'Hélène' },
  { id: 5, name: 'atlas', env: 'neo', pods: 7, owner: 'Zoé' },
];
const envs = [{ label: 'recette', value: 'recette' }, { label: 'neo', value: 'neo' }];
const COLUMNS: PoadTableColumn<Project>[] = [
  { field: 'name', header: 'Name', sortable: true, filterable: true },
  { field: 'env', header: 'Env', filterable: true, filterType: 'select', filterOptions: envs },
  { field: 'pods', header: 'Pods', dataType: 'number', sortable: true, filterable: true },
  { field: 'owner', header: 'Owner' },
];

export default function TablePanelDemo() {
  const [columns, setColumns] = useState(COLUMNS);
  const [filters, setFilters] = useState<TableFilters>({});
  const [search, setSearch] = useState('');
  return (
    <>
      <div className="demo-wide" style={{ display: 'flex', alignItems: 'stretch' }}>
        <PoadTable value={projects} columns={columns} dataKey="id" title="Projects" showFilterRow={false} filters={filters} globalFilter={search}
          onFilterChange={(e) => { setFilters(e.filters); setSearch(e.globalFilter); }} style={{ flex: 1, minWidth: 0 }} />
        <PoadTablePanel columns={columns} onColumnsChange={setColumns} filters={filters} onFiltersChange={setFilters} globalFilter={search} onGlobalFilterChange={setSearch} />
      </div>
      <code>filters={Object.keys(filters).filter((k) => filters[k]).join(',') || 'none'}</code>
    </>
  );
}

// app/page.tsx — composant serveur : HTML rendu sur le serveur, hydraté au client
import TablePanelDemo from './table-panel-demo';

export default function Page() {
  return <TablePanelDemo />;
}

<poad-table-panel for="id-de-la-table"> (ou la propriété table) : comme en Angular, le panneau pose columns, filters et globalFilter sur sa table et se redessine à chaque rendu de celle-ci. À quoi ça sert

Pour une table large dont on choisit les colonnes et qu’on filtre sans la ligne de filtres. Le panneau agit sur la table (mêmes colonnes, mêmes filtres) : ce qui change d’un côté se voit de l’autre. Sous 768 px il recouvre la table, fermé par défaut. Les onglets Groupes et Valeurs (agrégats) sont décrits avec les groupes de lignes.

  • Rail en motif « tabs » de WAI-ARIA : role="tablist" vertical, une seule tabulation (l’onglet actif), aria-selected, aria-controls vers le role="tabpanel" nommé par son onglet.
  • Chaque colonne a sa case nommée par son libellé, la case « toutes » passe à l’état partiel ; l’en-tête de chaque filtre est un bouton aria-expanded.

Comment changer

Dans la feuille globale de l’application, après les feuilles de @poadio/ui-core : sur :root pour toute l’application, ou sur un conteneur pour une zone seulement. Une couche flottante (menu, liste, dialogue) peut être rendue hors de son conteneur : pour elle, préférez :root.

src/styles.css
/* Toute l’application */
:root {
  --poad-accent: #0b7a5c;        /* Couleur d’accent : bouton principal, case cochée, sélection */
  --poad-radius: 4px;            /* Arrondi des coins des contrôles */
}

/* En thème sombre seulement */
.theme-dark {
  --poad-accent: #3fcf9f;        /* Couleur d’accent : bouton principal, case cochée, sélection */
}

/* Une zone seulement */
.checkout {
  --poad-radius: 0;              /* Arrondi des coins des contrôles */
}
