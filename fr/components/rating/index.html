Note

Une note en étoiles (ou toute icône du jeu), à la demi-étoile près si besoin ; le survol prévisualise, le clic choisit.

Démonstration

Rendu par Angular
import { Component, signal } from '@angular/core';
import { PoadRatingComponent } from '@poadio/ui-angular';

@Component({
  selector: 'app-rating-demo',
  imports: [PoadRatingComponent],
  template: `
    <poad-rating [(value)]="score" allowHalf showValue />
    <div class="demo-row">
      <poad-rating [value]="4" readonly size="sm" ariaLabel="Average rating" />
      <poad-rating [(value)]="likes" icon="heart" severity="danger" [max]="3" ariaLabel="Likes" />
    </div>
    <code>score={{ score() }} · likes={{ likes() }}</code>
  `,
})
export default class RatingDemo {
  readonly score = signal<number | null>(3.5);
  readonly likes = signal<number | null>(2);
}
Rendu par React (@poadio/ui-react)

Next.js emploie le même paquet que React : la démo est rendue par @poadio/ui-react. Le code montre l'intégration App Router — un composant client ('use client') pour l'état, rendu par une page serveur.

// app/rating-demo.tsx — composant client (état, gestionnaires)
'use client';

import { useState } from 'react';
import { PoadRating } from '@poadio/ui-react';

export default function RatingDemo() {
  const [score, setScore] = useState<number | null>(3.5);
  const [likes, setLikes] = useState<number | null>(2);
  return (
    <>
      <PoadRating value={score} onValueChange={setScore} allowHalf showValue />
      <div className="demo-row">
        <PoadRating defaultValue={4} readOnly size="sm" aria-label="Average rating" />
        <PoadRating value={likes} onValueChange={setLikes} icon="heart" severity="danger" max={3} aria-label="Likes" />
      </div>
      <code>score={String(score)} · likes={String(likes)}</code>
    </>
  );
}

// app/page.tsx — composant serveur : HTML rendu sur le serveur, hydraté au client
import RatingDemo from './rating-demo';

export default function Page() {
  return <RatingDemo />;
}

Rendu par Vue
<script setup lang="ts">
import { ref } from 'vue';
import { PoadRating } from '@poadio/ui-vue';

const score = ref<number | null>(3.5);
const likes = ref<number | null>(2);
</script>

<template>
  <PoadRating v-model="score" allow-half show-value />
  <div class="demo-row">
    <PoadRating :model-value="4" readonly size="sm" aria-label="Average rating" />
    <PoadRating v-model="likes" icon="heart" severity="danger" :max="3" aria-label="Likes" />
  </div>
  <code>score={{ score }} · likes={{ likes }}</code>
</template>

<script type="module">import '@poadio/ui-elements/define';</script>

<poad-rating id="score" name="score" value="3.5" allow-half show-value></poad-rating>
<div class="demo-row">
  <poad-rating value="4" readonly size="sm" aria-label="Average rating"></poad-rating>
  <poad-rating value="2" icon="heart" severity="danger" max="3" aria-label="Likes"></poad-rating>
</div>
<code id="score-out">score=3.5</code>

<script type="module">
  document.querySelector('#score').addEventListener('change', (e) => {

  });
</script>

Accessibilité

  • role="slider" focalisable avec aria-valuenow, aria-valuetext (« 2 / 5 »), aria-valuemin / aria-valuemax.
  • Nommée « Note » par le catalogue, ou par aria-label.
  • Lecture seule : aria-readonly="true", tabindex="-1", aucun geste. Miroir complet de droite à gauche.
