Next.js emploie le même paquet que React : la démo est rendue par @poadio/ui-react. Le code montre l'intégration App Router — un composant client ('use client') pour l'état, rendu par une page serveur.

PoadSwitch a un état : il commence par 'use client' dans le paquet. Une page serveur peut l'importer et le rendre ; seul le composant qui tient l'état (useState) doit être un composant client.

// app/switch-demo.tsx — composant client (état, gestionnaires)
'use client';

import { useState } from 'react';
import { PoadSwitch } from '@poadio/ui-react';

export default function SwitchDemo() {
  const [wifi, setWifi] = useState(true);
  const [bluetooth, setBluetooth] = useState(false);
  return (
    <>
      <div className="demo-row">
        <PoadSwitch label="Wi-Fi" checked={wifi} onCheckedChange={setWifi} />
        <PoadSwitch label="Bluetooth" labelPosition="left" checked={bluetooth} onCheckedChange={setBluetooth} />
        <PoadSwitch label="Disabled" defaultChecked disabled />
        <PoadSwitch label="Read-only" defaultChecked readOnly />
        <PoadSwitch aria-label="Wi-Fi (small)" size="sm" checked={wifi} onCheckedChange={setWifi} />
      </div>
      <code>wifi={String(wifi)} · bluetooth={String(bluetooth)}</code>
    </>
  );
}

// app/page.tsx — composant serveur : HTML rendu sur le serveur, hydraté au client
import SwitchDemo from './switch-demo';

export default function Page() {
  return <SwitchDemo />;
}

'use client';

import { useState } from 'react';
import { PoadCheckbox } from '@poadio/ui-react';

export default function CheckboxDemo() {
  const [postgis, setPostgis] = useState(true);

  return (
    <>

        <PoadCheckbox label="PostGIS" checked={postgis} onCheckedChange={setPostgis} />
        <PoadCheckbox label="pgvector" checked={vector} onCheckedChange={setVector} />
      </div>

Comment changer

Dans la feuille globale de l’application, après les feuilles de @poadio/ui-core : sur :root pour toute l’application, ou sur un conteneur pour une zone seulement. Une couche flottante (menu, liste, dialogue) peut être rendue hors de son conteneur : pour elle, préférez :root.

src/styles.css
/* Toute l’application */
:root {
  --poad-accent: #0b7a5c;        /* Couleur d’accent : bouton principal, case cochée, sélection */
  --poad-radius: 4px;            /* Arrondi des coins des contrôles */
}

/* En thème sombre seulement */
.theme-dark {
  --poad-accent: #3fcf9f;        /* Couleur d’accent : bouton principal, case cochée, sélection */
}

/* Une zone seulement */
.checkout {
  --poad-radius: 0;              /* Arrondi des coins des contrôles */
}
