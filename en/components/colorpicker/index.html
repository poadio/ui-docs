Color picker

<poad-colorpicker>

Picks a color: saturation / brightness area, hue and opacity with the pointer or the keyboard, HEX and RGB fields, palette, browser eyedropper. Value as CSS text.

// app/colorpicker-demo.tsx — client component (state, handlers)
'use client';

import { useState } from 'react';
import { PoadColorpicker } from '@poadio/ui-react';

export default function ColorpickerDemo() {
  const [accent, setAccent] = useState<string | null>('#2f6feb');
  const [overlay, setOverlay] = useState<string | null>('rgba(15, 23, 42, 0.6)');
  return (
    <>
      <div className="demo-row">
        <PoadColorpicker value={accent} onChange={setAccent} aria-label="Accent" clearable style={{ width: 200 }} />

      </div>
      <code>accent={String(accent)} · overlay={String(overlay)}</code>
    </>
  );
}

How to change

In the application’s global stylesheet, after the @poadio/ui-core sheets: on :root for the whole application, or on a container for one area only. A floating layer (menu, list, dialog) may be rendered outside its container: for it, prefer :root.

src/styles.css
/* The whole application */
:root {
  --poad-accent: #0b7a5c;        /* Accent colour: primary button, checked box, selection */
  --poad-radius: 4px;            /* Corner rounding of controls */
}

/* In dark theme only */
.theme-dark {
  --poad-accent: #3fcf9f;        /* Accent colour: primary button, checked box, selection */
}

/* One area only */
.checkout {
  --poad-radius: 0;              /* Corner rounding of controls */
}
