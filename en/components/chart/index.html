      <poad-chart [type]="type()" [data]="data" ariaLabel="Readings per month" tableToggle height="260px" (chartClick)="pick($event)" />
    </div>

  `,
})
export default class ChartDemo {
  readonly types = [{ label: 'Bars', value: 'bar' }, { label: 'Lines', value: 'line' }, { label: 'Doughnut', value: 'doughnut' }];
  readonly type = signal<PoadChartType>('bar');

    datasets: [{ label: 'Inspections', data: [12, 19, 8, 15, 11, 17] }, { label: 'Alerts', data: [3, 5, 2, 6, 4, 3] }],
  };
  readonly last = signal('—');
  pick(e: PoadChartClickEvent): void { this.last.set(`${e.label} ${e.value}`); }
}

// app/chart-demo.tsx — client component (state, handlers)
'use client';

import { useState } from 'react';
import { PoadChart, PoadSegmented, type PoadChartType } from '@poadio/ui-react';

const types = [{ label: 'Bars', value: 'bar' }, { label: 'Lines', value: 'line' }, { label: 'Doughnut', value: 'doughnut' }];
const data = {
  labels: ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun'],
  datasets: [{ label: 'Inspections', data: [12, 19, 8, 15, 11, 17] }, { label: 'Alerts', data: [3, 5, 2, 6, 4, 3] }],
};

export default function ChartDemo() {
  const [type, setType] = useState<PoadChartType>('bar');
  const [last, setLast] = useState('—');
  return (
    <>
      <PoadSegmented options={types} value={type} onValueChange={(v) => setType(v as PoadChartType)} size="sm" aria-label="Chart type" />
      <div className="demo-wide">
        <PoadChart type={type} data={data} ariaLabel="Readings per month" tableToggle height="260px" onChartClick={(e) => setLast(`${e.label} ${e.value}`)} />
      </div>
      <code>last={last}</code>
    </>
  );
}

What it is for

chart.js is an OPTIONAL dependency: npm i chart.js; it is only loaded when the first chart mounts (never during server rendering), and if it is missing the component says which package to install instead of breaking the page. Data has the Chart.js shape (labels, datasets); the options you pass are merged on top of the theme. For a single figure, a meter or a tag is enough.

Binding to a store

Declare the store once, then give it to the component: store (the object, or its name) and field or fields. Reading and writing go through the signed-in user’s RLS. All the details: Stores (@poadio/data).

import { store } from '@poadio/data';

// a Postgres function: one row per city
const stats = store.pgfn('stats', 'stats_villes');
Angular
<!-- stats: the store (component property) or its name -->
<poad-chart type="bar" [store]="stats" [fields]="[
  { field: 'ville', roles: ['label'] },
  { field: 'n', roles: ['value'] }
]" />
React · Next.js
<PoadChart type="bar" store={stats} fields={[
  { field: 'ville', roles: ['label'] },
  { field: 'n', roles: ['value'] }
]} />
Vue · Nuxt
<PoadChart type="bar" :store="stats" :fields="[
  { field: 'ville', roles: ['label'] },
  { field: 'n', roles: ['value'] }
]" />
Vanilla
<!-- the store by its name, declared before the components load (or el.store = the object) -->
<poad-chart type="bar" store="stats" fields='[
  { "field": "ville", "roles": ["label"] },
  { "field": "n", "roles": ["value"] }
]'></poad-chart>

Keyboard

KeyEffect
TabReaches the “Show data table” button (tableToggle). The canvas itself is not focusable: the table carries the values.
Enter SpaceOn the button: shows the data table in place of the drawing, then back to the chart.

Accessibility

  • A role="img" canvas named by ariaLabel: a screen reader announces what the chart represents.
  • A data table ALWAYS in the DOM (accessibleTable, on by default): caption = the chart’s name, one column per label (th scope="col"), one row per series (th scope="row"), numbers in the language. Visually hidden, it is still read by assistive technologies; showTable or the button (tableToggle) shows it in place of the canvas, for everyone.
  • Colours re-read from the tokens when the page changes theme (light / dark); a default palette contrasted on both backgrounds.

How to change

In the application’s global stylesheet, after the @poadio/ui-core sheets: on :root for the whole application, or on a container for one area only. A floating layer (menu, list, dialog) may be rendered outside its container: for it, prefer :root.

src/styles.css
/* The whole application */
:root {
  --poad-accent: #0b7a5c;        /* Accent colour: primary button, checked box, selection */
  --poad-radius-sm: 2px;         /* Small rounding: list options, small elements */
}

/* In dark theme only */
.theme-dark {
  --poad-accent: #3fcf9f;        /* Accent colour: primary button, checked box, selection */
}

/* One area only */
.checkout {
  --poad-radius-sm: 0;           /* Small rounding: list options, small elements */
}
