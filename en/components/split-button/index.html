Split button

<poad-split-button>

A main action and, joined to it, a chevron that opens a menu of variants (“Save and close”, “Draft”). Same looks as the button.

The menu is rendered in the CDK overlay, anchored on the chevron.

import { Component, signal } from '@angular/core';
import { PoadSplitButtonComponent, type PoadMenuItem } from '@poadio/ui-angular';

@Component({
  selector: 'app-split-button-demo',
  imports: [PoadSplitButtonComponent],
  template: `
    <div class="demo-row">
      <poad-split-button label="Save" icon="save" [items]="items" dropdownAriaLabel="More save options"
        (clicked)="saves.set(saves() + 1)" (itemClick)="last.set($event.item.label ?? '')" />
    </div>
    <code>saves={{ saves() }} · last={{ last() }}</code>
  `,
})
export default class SplitButtonDemo {
  readonly items: PoadMenuItem[] = [{ label: 'Save and close' }, { label: 'Save as draft' }];
  readonly saves = signal(0);
  readonly last = signal('none');
}

Next.js uses the same package as React: the demo is rendered by @poadio/ui-react. The code shows the App Router integration — a client component ('use client') for state, rendered by a server page.

// app/split-button-demo.tsx — client component (state, handlers)
'use client';

import { useState } from 'react';
import { PoadSplitButton, type PoadMenuItem } from '@poadio/ui-react';

const items: PoadMenuItem[] = [{ label: 'Save and close' }, { label: 'Save as draft' }];

export default function SplitButtonDemo() {
  const [saves, setSaves] = useState(0);
  const [last, setLast] = useState('none');
  return (
    <>
      <div className="demo-row">
        <PoadSplitButton
          label="Save" icon="save" items={items} dropdownAriaLabel="More save options"
          onClick={() => setSaves(saves + 1)} onItemClick={({ item }) => setLast(item.label ?? '')}
        />
      </div>
      <code>saves={saves} · last={last}</code>
    </>
  );
}

// app/page.tsx — server component: HTML rendered on the server, hydrated on the client
import SplitButtonDemo from './split-button-demo';

export default function Page() {
  return <SplitButtonDemo />;
}

<script type="module">
  const save = document.querySelector('#save');
  let saves = 0, last = 'none';
  const show = () => { document.querySelector('#split-out').textContent = `saves=${saves} · last=${last}`; };
  save.addEventListener('press', () => { saves++; show(); });
  save.addEventListener('itemclick', (e) => { last = e.detail.item.label; show(); });
</script>

What it is for

Accessibility

  • Two real buttons; the chevron has aria-haspopup="menu", aria-expanded and a name (the catalogue by default).
  • The menu is the Menu component: role="menu", role="menuitem" items, roving focus.

How to change

In the application’s global stylesheet, after the @poadio/ui-core sheets: on :root for the whole application, or on a container for one area only. A floating layer (menu, list, dialog) may be rendered outside its container: for it, prefer :root.

src/styles.css
/* The whole application */
:root {
  --poad-accent: #0b7a5c;        /* Accent colour: primary button, checked box, selection */
  --poad-radius: 4px;            /* Corner rounding of controls */
}

/* In dark theme only */
.theme-dark {
  --poad-accent: #3fcf9f;        /* Accent colour: primary button, checked box, selection */
}

/* One area only */
.checkout {
  --poad-radius: 0;              /* Corner rounding of controls */
}
