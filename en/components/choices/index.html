Next.js uses the same package as React: the demo is rendered by @poadio/ui-react. The code shows the App Router integration — a client component ('use client') for state, rendered by a server page.

PoadSwitch holds state: it starts with 'use client' in the package. A server page can import and render it; only the component that holds the state (useState) has to be a client component.

// app/switch-demo.tsx — client component (state, handlers)
'use client';

import { useState } from 'react';
import { PoadSwitch } from '@poadio/ui-react';

export default function SwitchDemo() {
  const [wifi, setWifi] = useState(true);
  const [bluetooth, setBluetooth] = useState(false);
  return (
    <>
      <div className="demo-row">
        <PoadSwitch label="Wi-Fi" checked={wifi} onCheckedChange={setWifi} />
        <PoadSwitch label="Bluetooth" labelPosition="left" checked={bluetooth} onCheckedChange={setBluetooth} />
        <PoadSwitch label="Disabled" defaultChecked disabled />
        <PoadSwitch label="Read-only" defaultChecked readOnly />
        <PoadSwitch aria-label="Wi-Fi (small)" size="sm" checked={wifi} onCheckedChange={setWifi} />
      </div>
      <code>wifi={String(wifi)} · bluetooth={String(bluetooth)}</code>
    </>
  );
}

    this.postgis.set(on);
    this.vector.set(on);
  }
}
