import { Component, signal } from '@angular/core';
import { PoadChipComponent } from '@poadio/ui-angular';

@Component({
  selector: 'app-chip-demo',
  imports: [PoadChipComponent],
  template: `
    <div class="demo-row">
      @for (city of cities(); track city) {
        <poad-chip [label]="city" icon="map-pin" removable (removed)="remove(city)" />
      }
    </div>
    <div class="demo-row">
      <poad-chip label="Mine only" clickable [selected]="mine()" (clicked)="mine.set(!mine())" />
      <poad-chip label="Ada Lovelace" avatarText="AL" severity="info" />
      <poad-chip label="Archived" size="sm" disabled removable />
    </div>
    <code>cities={{ cities().join(', ') }} · mine={{ mine() }}</code>
  `,
})
export default class ChipDemo {
  readonly cities = signal(['Paris', 'Lyon', 'Nantes']);
  readonly mine = signal(false);
  remove(city: string): void {
    this.cities.set(this.cities().filter((c) => c !== city));
  }
}

'use client';

In the application’s global stylesheet, after the @poadio/ui-core sheets: on :root for the whole application, or on a container for one area only. A floating layer (menu, list, dialog) may be rendered outside its container: for it, prefer :root.
