Checkbox

Checks or unchecks an option, with a partial state (aria-checked="mixed") for an “all” box when only some of its children are checked.

  • Angular
  • Next.js
  • React
  • Vue
  • Vanilla

Demo

Rendered by Angular
import { Component, computed, signal } from '@angular/core';
import { PoadCheckboxComponent } from '@poadio/ui-angular';

@Component({
  selector: 'app-checkbox-demo',
  imports: [PoadCheckboxComponent],
  template: `
    <div class="demo-row">
      <poad-checkbox [value]="all()" [indeterminate]="some()" label="All extensions" (changed)="setAll($event)" />
      <poad-checkbox [(value)]="postgis" label="PostGIS" />
      <poad-checkbox [(value)]="vector" label="pgvector" />
    </div>
    <div class="demo-row">
      <poad-checkbox [value]="true" label="Disabled" disabled />
      <poad-checkbox [value]="true" label="Read-only" readonly />
      <poad-checkbox [value]="false" label="Accept the terms" invalid />
    </div>
    <code>postgis={{ postgis() }} · pgvector={{ vector() }}</code>
  `,
})
export default class CheckboxDemo {
  readonly postgis = signal<boolean | null>(true);
  readonly vector = signal<boolean | null>(false);
  readonly all = computed(() => !!this.postgis() && !!this.vector());
  readonly some = computed(() => (!!this.postgis() || !!this.vector()) && !this.all());
  setAll(on: boolean): void {
    this.postgis.set(on);
    this.vector.set(on);
  }
}
Rendered by React (@poadio/ui-react)

Next.js uses the same package as React: the demo is rendered by @poadio/ui-react. The code shows the App Router integration — a client component ('use client') for state, rendered by a server page.

// app/checkbox-demo.tsx — client component (state, handlers)
'use client';

import { useState } from 'react';
import { PoadCheckbox } from '@poadio/ui-react';

export default function CheckboxDemo() {
  const [postgis, setPostgis] = useState(true);
  const [vector, setVector] = useState(false);
  const all = postgis && vector;
  const some = (postgis || vector) && !all;
  const setAll = (on: boolean) => { setPostgis(on); setVector(on); };
  return (
    <>
      <div className="demo-row">
        <PoadCheckbox label="All extensions" checked={all} indeterminate={some} onCheckedChange={setAll} />
        <PoadCheckbox label="PostGIS" checked={postgis} onCheckedChange={setPostgis} />
        <PoadCheckbox label="pgvector" checked={vector} onCheckedChange={setVector} />
      </div>
      <div className="demo-row">
        <PoadCheckbox label="Disabled" defaultChecked disabled />
        <PoadCheckbox label="Read-only" defaultChecked readOnly />
        <PoadCheckbox label="Accept the terms" invalid />
      </div>
      <code>postgis={String(postgis)} · pgvector={String(vector)}</code>
    </>
  );
}

// app/page.tsx — server component: HTML rendered on the server, hydrated on the client
import CheckboxDemo from './checkbox-demo';

export default function Page() {
  return <CheckboxDemo />;
}

What it is for

Use it for a choice submitted with the form, or for several independent options. For a setting that applies immediately, prefer a switch.

Keyboard

KeyEffect
SpaceChecks or unchecks (unless disabled or read-only); from the partial state, checks.

Accessibility

  • role="checkbox" and aria-checked (true, false or mixed) on a native <button>.
  • Named by its <label for>, otherwise by aria-label.
  • aria-invalid="true" when invalid; disabled: native disabled; read-only: focusable, frozen.
